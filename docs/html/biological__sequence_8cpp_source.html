32static string trim(const string &s) {
33 size_t start = 0;
34 size_t end = s.size();
35
36 while (start < end && s[start] <= ' ') {
37 ++start;
38 }
39
40 while (end > start && s[end - 1] <= ' ') {
41 --end;
42 }
43
44 return s.substr(start, end - start);
