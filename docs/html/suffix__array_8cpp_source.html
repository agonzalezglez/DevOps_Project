45static int _compareStrings(const string &s1, size_t p1,
46 const string &s2, size_t p2) {
47 int cmp = 0;
48 int lcp = 1;
49 size_t n1 = s1.length();
50 size_t n2 = s2.length();
51
52 while (!cmp && (p1 < n1) && (p2 < n2)) {
53 if (s1[p1] < s2[p2]) {
54 cmp = -lcp;
55 } else if (s1[p1] > s2[p2]) {
56 cmp = lcp;
57 } else {
58 ++p1;
59 ++p2;
60 ++lcp;
61 }
62 }
63

65 if (p1 < n1) {
66 cmp = lcp;
67 } else if (p2 < n2) {
68 cmp = -lcp;
69 }
70 }
