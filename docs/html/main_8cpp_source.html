90static vector<string> listSequenceFiles(const string &path) {

96
97 if (fs::is_regular_file(path)) {

99 files.push_back(path);
100 }
101 return files;
102 }
103
104 if (fs::is_directory(path)) {
105 for (fs::directory_iterator it(path); it != fs::directory_iterator(); ++it) {
106 string current = it->path().string();
107
108 if (fs::is_regular_file(it->path()) &&
109 hasExtension(current, {".fasta", ".fa", ".fna", ".fastq", ".fq"})) {
110 files.push_back(current);
111 }
112 }
113 }
