121vector<string> Mapper::extractKmers(const string &sequence, int k) {
122 vector<string> kmers;
123 string normalized = normalizeSequence(sequence);
124
125 if (k <= 0) {
126 return kmers;
127 }
128
129 if (normalized.size() < static_cast<size_t>(k)) {
130 return kmers;
131 }
132
133 for (size_t i = 0; i <= normalized.size() - static_cast<size_t>(k); ++i) {

260double Mapper::calculateMappingScore(int matchingKmers,
261 int totalKmers,
262 int mismatches) {
263 if (totalKmers <= 0) {
264 return 0.0;
265 }
266
267 double support = static_cast<double>(matchingKmers) / static_cast<double>(totalKmers);
268 double penalty = static_cast<double>(mismatches) / static_cast<double>(totalKmers);
269
270 double score = support - penalty;

360vector<Mapper::MappingResult> Mapper::mapRead(const BiologicalSequence &read,
361 const ReferenceIndex &index,
362 int maxMismatches) {
363 vector<MappingResult> results;
364
365 const string reference = normalizeSequence(index.reference());
366 const int k = index.k();
367
368 string readForward = normalizeSequence(read.getSequence());
369 string readReverse = getReverseComplement(readForward);
370
371 vector<pair<string, bool>> readVersions;
372 readVersions.push_back(make_pair(readForward, false));
373 readVersions.push_back(make_pair(readReverse, true));
374
375 for (size_t versionIndex = 0; versionIndex < readVersions.size(); ++versionIndex) {
376 const string &currentRead = readVersions[versionIndex].first;
377 bool isReverse = readVersions[versionIndex].second;
378
379 vector<string> kmers = extractKmers(currentRead, k);
380 if (kmers.empty()) {
381 continue;
382 }
383
392 unordered_map<int, int> positionCounts;
393
394 for (size_t i = 0; i < kmers.size(); ++i) {
395 vector<int> positions = index.getPositions(kmers[i]);
396
397 for (size_t j = 0; j < positions.size(); ++j) {
398 int candidateStart = positions[j] - static_cast<int>(i);
399
400 if (candidateStart < 0) {
401 continue;
402 }
403
404 positionCounts[candidateStart]++;
405 }
406 }
407
413 int minSupport = max(1, static_cast<int>(kmers.size() / 2));
414
415 for (unordered_map<int, int>::const_iterator it = positionCounts.begin();
416 it != positionCounts.end();
417 ++it) {
418 int candidatePosition = it->first;
419 int support = it->second;
420
421 if (support < minSupport) {
422 continue;
423 }
424
425 int mismatches = countMismatches(currentRead, reference, candidatePosition);
426
427 if (mismatches <= maxMismatches) {
428 MappingResult result;
429 result.position = candidatePosition;
430 result.isReverse = isReverse;
431 result.mismatches = mismatches;
432 result.matchedKmers = support;
433 result.totalKmers = static_cast<int>(kmers.size());
434 result.score = calculateMappingScore(support,
435 static_cast<int>(kmers.size()),
436 mismatches);
437 result.cigar = generateCigar(currentRead, reference, candidatePosition);
438
439 results.push_back(result);
440 }
441 }
442 }
443
444 sort(results.begin(), results.end(),
445 [](const MappingResult &a, const MappingResult &b) {
446 return a.score > b.score;
447 });
448
449 return results;
450}
