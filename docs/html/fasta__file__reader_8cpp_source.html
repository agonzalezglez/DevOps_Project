34static string trim_fasta(const string &s) {
35 size_t start = 0;
36 while ((start < s.size()) && isspace(static_cast<unsigned char>(s[start]))) {
37 ++start;
38 }
39
40 size_t end = s.size();
41 while ((end > start) && isspace(static_cast<unsigned char>(s[end - 1]))) {
42 --end;
43 }
44
45 return s.substr(start, end - start);
